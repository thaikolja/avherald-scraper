This page is copied from the repository `config.ini` when the documentation is built. Command-line flags still override these values.

```ini [config.ini]
; =============================================================================
;  The Aviation Herald Scraper
; =============================================================================
;
;  Default settings for this project. Edit the values in place.
;  When a setting is also a command-line flag, the flag wins.
;
;  Booleans are true or false.
;  Category lists are comma-separated names.

[site]

; Public origin. Article and list URLs are built from this.
base_url = https://avherald.com

[storage]

; SQLite file. A relative path starts from the directory you run the command in.
; Resume always uses this file.
database = ./data/avherald.sqlite

; Output type: sqlite, json, csv, xlsx, cli.
; cli prints the rows in the terminal. Use it for a few items.
; json, csv, and xlsx are written next to the database, with the same name.
format = sqlite

[crawl]

; Unique articles to keep. IDs already in the database do not count.
limit = 10

; List order on the site: update or occurrence.
sort = update

; Comma-separated: crash, accident, incident, news, report.
; Leave this unset to keep every category.
; categories = incident,accident

; Stop after this many list pages. Leave this unset for no page cap.
; max_pages = 3

; Continue from the last saved list page.
resume = true

; Ignore the saved position and start from the first list page.
fresh = false

; Parse and print rows without writing the database.
dry_run = false

; Print each fetched URL and the kept rows.
verbose = false

[network]

; Seconds between list-page requests. The first request is not delayed.
delay = 3

; Per-request timeout, in seconds.
timeout = 20

; HTTP(S) proxies, comma-separated. A ban or a dead host tries the next one.
; proxy = http://127.0.0.1:8080, http://127.0.0.1:8081
```
